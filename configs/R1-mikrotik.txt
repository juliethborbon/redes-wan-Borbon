add dst-address=192.168.40.0/28 gateway=10.0.0.9 comment="LAN BODEGA de MK1"

add list=LAN-INTERNA address=192.168.10.0/26
add list=LAN-INTERNA address=192.168.10.64/27

add list=SEDES address=192.168.40.0/28
/ip firewall nat
add chain=srcnat out-interface=ether3 action=masquerade comment="NAT hacia el ISP"
/ip firewall filter
add chain=input connection-state=established,related action=accept comment="Respuestas"
add chain=input connection-state=invalid action=drop
add chain=input protocol=icmp action=accept comment="Ping"

add chain=forward src-address-list=SEDES dst-address-list=LAN-INTERNA action=accept comment="Otras sedes hacia la LAN"
add chain=forward src-address-list=SEDES out-interface=ether3 action=accept comment="Otras sedes hacia Internet"
add chain=forward action=drop comment="Bloquear el resto"
/ip service
disable telnet,ftp,api,api-ssl,www
